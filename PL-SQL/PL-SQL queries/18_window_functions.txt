SQL PART 18 — WINDOW FUNCTIONS
================================

Window functions calculate across related rows without
collapsing them into one row.

1. ROW_NUMBER:

SELECT
    name,
    marks,
    ROW_NUMBER() OVER (
        ORDER BY marks DESC
    ) AS row_num
FROM students;

OUTPUT:
Priya 91 1
Rahul 85 2
Rohit 78 3
Aman 72 4
Neha 68 5

2. RANK:

SELECT
    name,
    marks,
    RANK() OVER (
        ORDER BY marks DESC
    ) AS ranking
FROM students;

3. DENSE_RANK:

SELECT
    name,
    marks,
    DENSE_RANK() OVER (
        ORDER BY marks DESC
    ) AS ranking
FROM students;

Difference:
ROW_NUMBER: always unique sequence.
RANK: ties create gaps.
DENSE_RANK: ties do not create gaps.

4. PARTITION BY:

SELECT
    name,
    city,
    marks,
    RANK() OVER (
        PARTITION BY city
        ORDER BY marks DESC
    ) AS city_rank
FROM students;

Ranking restarts for every city.

5. LAG:

SELECT
    name,
    marks,
    LAG(marks) OVER (
        ORDER BY marks
    ) AS previous_marks
FROM students;

6. LEAD:

SELECT
    name,
    marks,
    LEAD(marks) OVER (
        ORDER BY marks
    ) AS next_marks
FROM students;

7. Running total:

SELECT
    name,
    marks,
    SUM(marks) OVER (
        ORDER BY student_id
    ) AS running_total
FROM students;
