SQL PART 2 — SELECT, WHERE & OPERATORS
========================================

SELECT * FROM students;

SELECT name, marks
FROM students;

WHERE:

SELECT *
FROM students
WHERE marks > 80;

AND:

SELECT *
FROM students
WHERE age >= 20 AND marks > 75;

OR:

SELECT *
FROM students
WHERE city = 'Delhi' OR city = 'Bhopal';

NOT:

SELECT *
FROM students
WHERE NOT city = 'Delhi';

BETWEEN:

SELECT *
FROM students
WHERE marks BETWEEN 70 AND 85;

IN:

SELECT *
FROM students
WHERE city IN ('Delhi', 'Bhopal', 'Lucknow');

LIKE:

SELECT *
FROM students
WHERE name LIKE 'R%';

SELECT *
FROM students
WHERE name LIKE '%a';
