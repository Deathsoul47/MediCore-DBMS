SQL PART 19 — ADVANCED QUERY PATTERNS
=======================================

1. TOP 2 STUDENTS PER CITY

WITH ranked_students AS (
    SELECT
        name,
        city,
        marks,
        ROW_NUMBER() OVER (
            PARTITION BY city
            ORDER BY marks DESC
        ) AS rn
    FROM students
)
SELECT
    name,
    city,
    marks
FROM ranked_students
WHERE rn <= 2;

This combines:
CTE + ROW_NUMBER + PARTITION BY.

2. SECOND HIGHEST MARKS

SELECT MAX(marks) AS second_highest
FROM students
WHERE marks < (
    SELECT MAX(marks)
    FROM students
);

3. NTH HIGHEST MARKS

SELECT marks
FROM (
    SELECT
        marks,
        DENSE_RANK() OVER (
            ORDER BY marks DESC
        ) AS rnk
    FROM students
) x
WHERE rnk = 3;

4. DUPLICATE VALUES

SELECT city, COUNT(*) AS total
FROM students
GROUP BY city
HAVING COUNT(*) > 1;

5. EXISTS

SELECT *
FROM students s
WHERE EXISTS (
    SELECT 1
    FROM students3 s3
    WHERE s3.student_id = s.student_id
);

6. NOT EXISTS

SELECT *
FROM departments d
WHERE NOT EXISTS (
    SELECT 1
    FROM students3 s
    WHERE s.department_id = d.department_id
);

7. CONDITIONAL AGGREGATION

SELECT
    COUNT(*) AS total,
    SUM(CASE WHEN marks >= 80 THEN 1 ELSE 0 END) AS above_80
FROM students;
