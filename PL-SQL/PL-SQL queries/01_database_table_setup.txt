SQL PART 1 — DATABASE & TABLE SETUP
====================================

1. Create database

CREATE DATABASE college_db;
USE college_db;

2. Create students table

CREATE TABLE students (
    student_id INT PRIMARY KEY,
    name VARCHAR(50),
    age INT,
    gender VARCHAR(10),
    city VARCHAR(50),
    marks INT
);

3. Insert sample data

INSERT INTO students
(student_id, name, age, gender, city, marks)
